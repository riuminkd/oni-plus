inf_calc_gene_pillars_level_up_check_fixed = {
	hidden_effect = {
		if = {
			limit = {
				check_variable = { which = aPUI_gene_pillar_xp_var which = aPUI_gene_pillar_xp_capacity }
			}
			subtract_variable = { which = aPUI_gene_pillar_xp_var which = aPUI_gene_pillar_xp_capacity }
			owner = {
				set_aPUI_var = { name = gene_pillar_xp which = PREV }
				change_aPUI_var = { name = gene_pillars_party_level value = 1 }

				if = {
					limit = { is_variable_equal = { which = aPUI_gene_pillars_party_level_var value = 2 } }
					gene_pillars = {
						limit = { owned_by = PREV }
						set_variable = { which = aPUI_gene_pillar_party_slot_framevar_2 value = 1 }
					}
					set_aPUI_cap = { name = gene_pillar_xp amount = 100 }
				}
				else_if = {
					limit = { is_variable_equal = { which = aPUI_gene_pillars_party_level_var value = 3 } }
					gene_pillars = {
						limit = { owned_by = PREV }
						set_variable = { which = aPUI_gene_pillar_party_slot_framevar_3 value = 1 }
					}
					set_aPUI_cap = { name = gene_pillar_xp amount = 100 }

					if = {
						limit = { NOT = { has_country_modifier = mechanim_administration } }
						set_country_flag = gene_pillars_unlocked_fighter_B
						set_country_flag = gene_pillars_unlocked_fighter_C
					}
				}
				else_if = {
					limit = { is_variable_equal = { which = aPUI_gene_pillars_party_level_var value = 4 } }
					gene_pillars = {
						limit = { owned_by = PREV }
						set_variable = { which = aPUI_gene_pillar_party_slot_framevar_4 value = 1 }
					}
					set_aPUI_cap = { name = gene_pillar_xp amount = 200 }

					if = {
						limit = { NOT = { has_country_modifier = mechanim_administration } }
						set_country_flag = gene_pillars_unlocked_cleric_B
					}
				}
				else_if = {
					limit = { is_variable_equal = { which = aPUI_gene_pillars_party_level_var value = 5 } }
					gene_pillars = {
						limit = { owned_by = PREV }
						set_variable = { which = aPUI_gene_pillar_party_slot_framevar_5 value = 1 }
					}
					set_variable = { which = aPUI_gene_pillar_xp_var value = 200 }
				}
				gene_pillars = {
					limit = { owned_by = PREV }
					set_aPUI_var = { name = gene_pillars_party_level which = owner }
				}
			}
			set_aPUI_var = { name = gene_pillar_xp which = owner }
			set_aPUI_cap = { name = gene_pillar_xp which = owner }
			set_aPUI_var = { name = gene_pillars_party_level which = owner }
			update_aPUI_var_bar = { name = gene_pillar_xp }
		}
	}
}

inf_calc_gene_pillars_finish_spirit_6950_level_3 = {
	hidden_effect = {
		6950 = {
			save_event_target_as = gene_pillar_exploration_province
		}
	}

	6950 = {
		remove_province_modifier = gene_pillars_spirit_return_of_tower_master
	}

	if = {
		limit = { is_variable_equal = { which = gene_pillars_spirit_lock_power value = 4 } }
		gene_pillars_add_loot_effect = { amount = 25 }
		gene_pillars_add_xp_effect = { amount = 5 }
	}
	else_if = {
		limit = { is_variable_equal = { which = gene_pillars_spirit_lock_power value = 3 } }
		gene_pillars_add_loot_effect = { amount = 20 }
		gene_pillars_add_xp_effect = { amount = 4 }
		6950 = {
			add_province_modifier = {
				name = gene_pillars_spirit_whispers
				duration = -1
			}
		}
	}
	else_if = {
		limit = { is_variable_equal = { which = gene_pillars_spirit_lock_power value = 2 } }
		gene_pillars_add_loot_effect = { amount = 15 }
		gene_pillars_add_xp_effect = { amount = 3 }
		6950 = {
			add_province_modifier = {
				name = gene_pillars_spirit_haunting
				duration = -1
			}
		}
	}
	else = {
		gene_pillars_add_loot_effect = { amount = 10 }
		gene_pillars_add_xp_effect = { amount = 2 }
		6950 = {
			add_province_modifier = {
				name = gene_pillars_spirit_shattered
				duration = -1
			}
		}
	}

	hidden_effect = {
		set_country_flag = aPUI_gene_pillar_explored_level_3_in_6950
		clr_country_flag = gene_pillars_has_pending_event
		clr_country_flag = gene_pillars_additional_pending_event
	}
	gene_pillar_successful_end_expedition_effect = yes
}
